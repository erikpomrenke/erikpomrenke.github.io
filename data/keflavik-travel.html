def seasonality_index(nationality_name):
    sub = totals[totals['nationality'] == nationality_name]
    monthly_avg = sub.groupby('month')['passengers'].mean()
    return monthly_avg / monthly_avg.mean()

seasonality = pd.DataFrame({
    'Icelanders': seasonality_index('Ísland'),
    'Foreigners': seasonality_index('Útlendingar alls')
})

seasonality.plot(kind='bar', figsize=(10, 5))
plt.axhline(1.0, color='gray', linestyle='--', linewidth=1)
plt.title("Seasonality index by month (1.0 = average month)")
plt.ylabel("Ratio to yearly average")
plt.xlabel("Month")
plt.show()
